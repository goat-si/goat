No test reads saved pairing bytes: `rg -n 'baseline.*pairing|pairing.*baseline' packages/ssh/src --glob '*.ts'` returned no matches. tunnel.core.test.ts reads saved runner and launch bytes, then compares newly generated pairing variants to each other.

The following exact transformation was run from the lane root before committing #887. It independently generated the base and modified runner/launch/pairing in native Node and compared all nine entries, and compared each transformed saved runner/launch to the native generator; saved pairing was transformed without unrelated refresh. It restores modified sources in a finally block.

```python
import subprocess,json,pathlib
paths=[pathlib.Path('packages/ssh/src/remoteRunnerScript.ts'),pathlib.Path('packages/ssh/src/runtimeFactsScripts.ts')]
heads=[p.read_text() for p in paths]
script='''import * as t from './packages/ssh/src/tunnel.ts';const inputs={archive:{archiveVersion:'1.2.3-preview.20260911.4'},mirror:{archiveVersion:'1.2.3-preview.20260911.4',releaseBaseUrl:'https://example.invalid/mirror/'},node:{nodeScriptPath:'/fixture/cli entry.mjs',nodeEngineRange:'^22.16 || ^23.11 || >=24.10'}};const output={};for(const [key,input] of Object.entries(inputs)) output[key]={runner:t.buildRemoteGoatRunnerScript(input),launch:t.buildRemoteLaunchScript(input),pairing:t.buildRemotePairingScript('711bc738002d72fd',input)};console.log(JSON.stringify(output));'''
def generate(): return json.loads(subprocess.check_output(['/Users/jun/.local/share/vite-plus/js_runtime/node/26.12.1/bin/node','--input-type=module','-e',script],text=True,env={'PATH':'/usr/bin:/bin:/usr/sbin:/sbin'}))
try:
 for p in paths:p.write_text(subprocess.check_output(['git','show',f'HEAD:{p}'],text=True))
 before=generate()
finally:
 for p,s in zip(paths,heads):p.write_text(s)
after=generate()
changes=[('sha256sum "$GOAT_STAGING/$GOAT_ARCHIVE"','sha256sum < "$GOAT_STAGING/$GOAT_ARCHIVE"'),('shasum -a 256 "$GOAT_STAGING/$GOAT_ARCHIVE"','shasum -a 256 < "$GOAT_STAGING/$GOAT_ARCHIVE"'),('shasum -a 256 "$GOAT_STAGING/goat"','shasum -a 256 < "$GOAT_STAGING/goat"'),('sha256sum "$GOAT_STAGING/goat"','sha256sum < "$GOAT_STAGING/goat"'),('shasum -a 256 "$1"','shasum -a 256 < "$1"'),('sha256sum "$1"','sha256sum < "$1"')]
anchor='  # Publish provenance with the verified executable, never backfill legacy directories.\n'
check='  case "$GOAT_EXECUTABLE_SHA256" in *[!0123456789abcdef]*|\'\') printf \'Could not verify the GOAT executable checksum.\\n\' >&2; exit 1 ;; esac\n  [ "${#GOAT_EXECUTABLE_SHA256}" = 64 ] || { printf \'Could not verify the GOAT executable checksum.\\n\' >&2; exit 1; }\n'
def transform(s):
 for a,b in changes:s=s.replace(a,b)
 return s.replace(anchor,check+anchor)
fixture=pathlib.Path('packages/ssh/src/fixtures/remote-runner-baseline.json')
saved=json.loads(subprocess.check_output(['git','show',f'HEAD:{fixture}'],text=True))
for key in saved:
 for field in saved[key]:
  assert transform(before[key][field])==after[key][field],(key,field,'base-generated')
  result=transform(saved[key][field])
  if field!='pairing':assert result==after[key][field],(key,field,'fixture')
  saved[key][field]=result
fixture.write_text(json.dumps(saved,ensure_ascii=False,indent=2)+'\n')
print('PASS: nine native base/head generated comparisons; six fixture runner/launch comparisons; three pairing transformations')
```
