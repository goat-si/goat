const threadId = ThreadId.make("thread-setup-terminal");
const projectId = ProjectId.make("project-setup");

// A restore holds the worktree's lease and runs the project's setup script, whose terminal asks
// for the same lease (goat#3674). Real TerminalManager and setup runner; it.live because the
// restore and the fixtures run git as a subprocess.
type Gates = { beforeOpen?: Effect.Effect<void>; beforeSentinel?: Effect.Effect<void> };
const withSetupProject = <A, E, R>(
  setup: { readonly async: boolean; readonly gates?: Gates },
  body: (fixture: {
    readonly worktreePath: string;
    readonly pty: FakePtyAdapter;
    readonly dispatched: Array<{ readonly type: string; readonly worktreePath?: unknown }>;
    readonly git: (cwd: string, ...args: string[]) => string;
    readonly terminals: TerminalManager.TerminalManager["Service"];
    readonly layer: Layer.Layer<
      | ProjectService.ProjectService
      | ProjectSetupScriptRunner.ProjectSetupScriptRunner
      | ProjectionStore.ProjectionStoreV2
      | ThreadManagement.ThreadManagementService
      | WorktreeSnapshotStore,
      never,
      FileSystem.FileSystem | Path.Path | ServerSettings.ServerSettingsService
    >;
  }) => Effect.Effect<A, E, R>,
) =>
  Effect.gen(function* () {
    const config = yield* ServerConfig.ServerConfig;
    const root = NodeFS.realpathSync(config.baseDir);
    const repo = NodePath.join(root, "repo");
    const worktreePath = NodePath.join(config.worktreesDir, "repo", "feature");
    const git = (cwd: string, ...args: string[]) =>
      NodeChildProcess.execFileSync("git", args, { cwd, encoding: "utf8" }).trim();
    NodeFS.mkdirSync(repo, { recursive: true });
    git(repo, "init", "-q", "-b", "main");
    git(
      repo,
      "-c",
      "user.name=T",
      "-c",
      "user.email=t@example.com",
      "commit",
      "-q",
      "--allow-empty",
      "-m",
      "base",
    );
    git(repo, "worktree", "add", "-q", "-b", "feature", worktreePath);
    const project = {
      id: projectId,
      title: "Setup",
      workspaceRoot: repo,
      scripts: [
        {
          id: "setup",
          name: "Setup",
          command: "true",
          icon: "configure",
          runOnWorktreeCreate: true,
          stopOnSettle: false,
          async: setup.async,
        },
      ],
    };
    const pty = new FakePtyAdapter();
    const terminals = yield* TerminalManager.makeWithOptions({
      logsDir: NodePath.join(root, "terminal-logs"),
      ptyAdapter: pty,
      processKillGraceMs: 1,
    });
    const projects = Layer.succeed(ProjectService.ProjectService, {
      getById: () => Effect.succeedSome(project),
      getByWorkspaceRoot: () => Effect.succeedSome(project),
    } as unknown as ProjectService.ProjectService["Service"]);
    const setupRunner = Layer.effect(
      ProjectSetupScriptRunner.ProjectSetupScriptRunner,
      ProjectSetupScriptRunner.make,
    ).pipe(
      Layer.provide(
        Layer.mergeAll(
          projects,
          // A shell that shows its prompt once open and reports the setup command exited with 0.
          Layer.succeed(TerminalManager.TerminalManager, {
            ...terminals,
            open: (input) =>
              Effect.andThen(
                setup.gates?.beforeOpen ?? Effect.void,
                terminals.open(input).pipe(
                  Effect.tap(() =>
                    Effect.sync(() => pty.processes.at(-1)?.emitData("$ \x1b[?2004h")),
                  ),
                ),
              ),
            write: (input) =>
              terminals.write(input).pipe(
                Effect.tap(() => {
                    const sentinel = input.data.match(/__GOAT_SETUP_DONE__[a-f0-9_]+:/)?.[0];
                    if (sentinel === undefined) return Effect.void;
                    const proc = pty.processes.at(-1);
                    return (setup.gates?.beforeSentinel ?? Effect.void).pipe(Effect.andThen(Effect.sync(() => proc?.emitData(`ready\r\n${sentinel}0\r\n$ `))));
                }),
              ),
          }),
          GoatProjectFileLoader.layer,
          NodeCrypto.layer,
        ),
      ),
    );
    const dispatched: Array<{ readonly type: string; readonly worktreePath?: unknown }> = [];
    const layer = Layer.mergeAll(
      projects,
      setupRunner,
      Layer.succeed(ProjectionStore.ProjectionStoreV2, {
        getThreadShell: () =>
          Effect.succeed(shell({ id: threadId, projectId, branch: "feature", worktreePath })),
      } as unknown as ProjectionStore.ProjectionStoreV2["Service"]),
      Layer.succeed(ThreadManagement.ThreadManagementService, {
        dispatch: (command: { readonly type: string; readonly worktreePath?: unknown }) =>
          Effect.sync(() => dispatched.push(command)),
      } as unknown as ThreadManagement.ThreadManagementService["Service"]),
      Layer.succeed(WorktreeSnapshotStore, {
        capture: () => Effect.die("unused"),
        latestForThread: () => Effect.succeed(null),
        markRestored: () => Effect.void,
        deleteForThread: () => Effect.succeed(0),
        purgeDeletedThreads: () => Effect.succeed(0),
        expire: () => Effect.succeed(0),
      }),
    );
    return yield* body({ worktreePath, pty, dispatched, git, terminals, layer });
  }).pipe(
    Effect.provide(
      layerRealGit.pipe(
        Layer.provideMerge(
          Layer.mergeAll(
            ServerSettings.layerTest(),
            ServerConfig.layerTest(process.cwd(), { prefix: "goat-setup-terminal-" }),
            McpProviderSessions.layer,
            ProcessRunner.layer,
          ),
        ),
        Layer.provideMerge(NodeServices.layer),
      ),
    ),
    Effect.scoped,
  );

/** What a restore stopped by a restart leaves: the folder, marked until setup has run in it. */
const markUnfinished = (worktreePath: string, git: (cwd: string, ...args: string[]) => string) =>
  Effect.gen(function* () {
    const pending = yield* makeRestorePending;
    yield* pending.write(worktreePath, {
      restoredFrom: "branch",
      headSha: git(worktreePath, "rev-parse", "HEAD"),
      snapshotSha: null,
      snapshotRef: null,
    });
    return pending;
  });


const probe = (window: "beforeOpen" | "beforeSentinel", withPerson: boolean) =>
  Effect.gen(function* () {
    const reached = yield* Deferred.make<void>();
    const go = yield* Deferred.make<void>();
    const gate = Deferred.succeed(reached, undefined).pipe(Effect.andThen(Deferred.await(go)));
    return yield* withSetupProject({ async: false, gates: { [window]: gate } }, ({ worktreePath, pty, git, terminals, layer }) =>
      Effect.gen(function* () {
        const pending = yield* markUnfinished(worktreePath, git);
        const service = yield* ThreadWorkspace.ThreadWorkspaceService;
        const restore = yield* service.restore({ threadId, mode: "restore" }).pipe(Effect.forkChild);
        const gateHit = yield* Deferred.await(reached).pipe(Effect.timeoutOption("8 seconds"));
        if (Option.isNone(gateHit)) { console.log("PROBE gate not reached", window, JSON.stringify(yield* Fiber.join(restore).pipe(Effect.exit, Effect.timeoutOption("1 second")))); return { window, withPerson, restoreFinished: false, personFinished: null, spawned: [], marked: true }; }
        let person: Fiber.Fiber<unknown, unknown> | undefined;
        if (withPerson) {
          // A person opens another terminal in the same thread while the restore holds the folder.
          person = yield* terminals.open({ threadId, terminalId: "person", cwd: worktreePath, worktreePath } as never).pipe(Effect.forkChild);
          yield* Effect.sleep("300 millis");
        }
        yield* Deferred.succeed(go, undefined);
        const finished = yield* Fiber.join(restore).pipe(Effect.timeoutOption("6 seconds"));
        const personDone = person === undefined ? null : Option.isSome(yield* Fiber.join(person).pipe(Effect.exit, Effect.timeoutOption("1 second")));
        const out = { window, withPerson, restoreFinished: Option.isSome(finished), personFinished: personDone, spawned: pty.spawnInputs.map((i) => i.cwd === worktreePath ? "wt" : i.cwd), marked: yield* pending.marked(worktreePath) };
        console.log("PROBE", JSON.stringify(out));
        yield* Fiber.interrupt(restore).pipe(Effect.timeoutOption("2 seconds"));
        if (person) yield* Fiber.interrupt(person).pipe(Effect.timeoutOption("2 seconds"));
        return out;
      }).pipe(Effect.provide(ThreadWorkspace.layer.pipe(Layer.provideMerge(layer)))),
    );
  });

it.live("probe control beforeOpen", () => probe("beforeOpen", false).pipe(Effect.map((o) => assert.isTrue(o.restoreFinished))), 30_000);
it.live("probe person beforeOpen", () => probe("beforeOpen", true).pipe(Effect.map((o) => assert.isTrue(o.restoreFinished, JSON.stringify(o)))), 30_000);
it.live("probe control beforeSentinel", () => probe("beforeSentinel", false).pipe(Effect.map((o) => assert.isTrue(o.restoreFinished))), 30_000);
it.live("probe person beforeSentinel", () => probe("beforeSentinel", true).pipe(Effect.map((o) => assert.isTrue(o.restoreFinished, JSON.stringify(o)))), 30_000);
