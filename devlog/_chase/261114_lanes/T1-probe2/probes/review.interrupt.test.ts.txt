import * as NodeServices from "@goat/shared/testing/nodeServicesWithCrypto";
import { assert, it } from "@effect/vitest";
import * as McpProviderSessions from "@goat/provider-core/server/McpProviderSessions";
import type * as PtyAdapter from "@goat/shared/PtyAdapter";
import * as Deferred from "effect/Deferred";
import * as Effect from "effect/Effect";
import * as Fiber from "effect/Fiber";
import * as FileSystem from "effect/FileSystem";
import * as Layer from "effect/Layer";
import * as Option from "effect/Option";
import * as Path from "effect/Path";
import * as ProcessRunner from "../processRunner.ts";
import { tryWithWorkspaceLease, withWorkspaceLease } from "../workspace/workspaceLease.ts";
import { createManager, FakePtyAdapter, openInput } from "./Manager.testkit.ts";
class GatedPtyAdapter extends FakePtyAdapter {
  gate: { readonly spawning: Deferred.Deferred<void>; readonly release: Deferred.Deferred<void> } | null = null;
  override spawn(input: PtyAdapter.PtySpawnInput) {
    const gate = this.gate; this.gate = null;
    if (!gate) return super.spawn(input);
    return Deferred.succeed(gate.spawning, undefined).pipe(Effect.andThen(Deferred.await(gate.release)), Effect.andThen(super.spawn(input)));
  }
}
it.layer(Layer.mergeAll(NodeServices.layer, McpProviderSessions.layer, ProcessRunner.layer.pipe(Layer.provide(NodeServices.layer))), { excludeTestServices: true })("review interrupt probe", (it) => {
  for (const kind of ["attach", "open"] as const)
  it.effect("PROBE " + kind + " interrupted while its rerun holds the lease and waits for the thread lock", () =>
    Effect.gen(function* () {
      const ptyAdapter = new GatedPtyAdapter();
      const { manager } = yield* createManager(5, { ptyAdapter });
      const fs = yield* FileSystem.FileSystem; const { join } = yield* Path.Path;
      const base = yield* fs.makeTempDirectoryScoped({ prefix: "goat-review-int-" });
      const cwd = join(base, "held"); const other = join(base, "other");
      yield* fs.makeDirectory(cwd); yield* fs.makeDirectory(other);
      const held = yield* Deferred.make<void>(); const release = yield* Deferred.make<void>();
      const holder = yield* withWorkspaceLease(cwd, Deferred.succeed(held, undefined).pipe(Effect.andThen(Deferred.await(release)))).pipe(Effect.forkChild);
      yield* Deferred.await(held);
      const input = openInput({ cwd });
      const req = kind === "attach"
        ? yield* manager.attachStream(input, () => Effect.void).pipe(Effect.forkChild({ startImmediately: true }))
        : yield* manager.open(input).pipe(Effect.forkChild({ startImmediately: true }));
      const gate = { spawning: yield* Deferred.make<void>(), release: yield* Deferred.make<void>() };
      ptyAdapter.gate = gate;
      const otherOpen = yield* manager.open(openInput({ terminalId: "other", cwd: other })).pipe(Effect.forkChild({ startImmediately: true }));
      yield* Deferred.await(gate.spawning);
      yield* Deferred.succeed(release, undefined);
      yield* Fiber.join(holder);
      let spins = 0;
      while (Option.isSome(yield* tryWithWorkspaceLease(cwd, Effect.void)) && spins < 1000) { spins++; yield* Effect.yieldNow; }
      const leaseHeldByRerun = spins < 1000;
      yield* Fiber.interrupt(req);
      const leaseFreeAfter = Option.isSome(yield* tryWithWorkspaceLease(cwd, Effect.void));
      yield* Deferred.succeed(gate.release, undefined);
      yield* Fiber.join(otherOpen);
      const spawnedBeforeNext = ptyAdapter.spawnInputs.map((s) => (s.cwd === cwd ? "held" : "other"));
      yield* manager.open(input);
      const out = { kind, spins, leaseHeldByRerun, leaseFreeAfter, spawnedBeforeNext, spawnedAfter: ptyAdapter.spawnInputs.map((s) => (s.cwd === cwd ? "held" : "other")) };
      console.log("REVIEW-PROBE interrupt " + JSON.stringify(out));
      assert.isTrue(leaseHeldByRerun); assert.isTrue(leaseFreeAfter); assert.deepEqual(spawnedBeforeNext, ["other"]); assert.deepEqual(out.spawnedAfter, ["other", "held"]);
    }),
  );
});
